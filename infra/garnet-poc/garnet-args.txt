# ══════════════════════════════════════════════════════════════════════════════
# PRODUCTION ARGUMENTS (use with NSSM or sc.exe service wrapper)
# ══════════════════════════════════════════════════════════════════════════════

--bind 127.0.0.1
--port 6379
--auth <REPLACE_WITH_PRODUCTION_PASSWORD>
--checkpointdir C:\Garnet\data
--checkpoint-freq 300

# ══════════════════════════════════════════════════════════════════════════════
# EXPLANATION
# ══════════════════════════════════════════════════════════════════════════════

# --bind 127.0.0.1       : DEPLOY-08/09 — listen only on localhost, no external access
# --port 6379            : Standard Redis port — no app config change needed
# --auth <password>      : requirepass equivalent — must match ConnectionStrings:Redis in appsettings
# --checkpointdir <path> : RDB-equivalent persistence directory (DEPLOY-11)
# --checkpoint-freq 300  : Checkpoint every 5 minutes (300 seconds)

# ══════════════════════════════════════════════════════════════════════════════
# ADDITIONAL OPTIONS (optional, for tuning)
# ══════════════════════════════════════════════════════════════════════════════

# --memory <bytes>       : Memory limit (e.g., --memory 1073741824 for 1GB)
# --threads <n>          : Number of worker threads (default: auto)
# --aof                  : Enable AOF persistence (in addition to checkpoints)
# --cluster              : Enable cluster mode (not needed for single-node)
