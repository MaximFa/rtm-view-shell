& {
  function Tick($text) { Write-Host ("[{0}] {1}" -f (Get-Date -Format 'HH:mm:ss'), $text) -ForegroundColor Cyan }
  if ($env:COMPUTERNAME -ne 'RTM') { Write-Host 'STOP: not server 234' -ForegroundColor Red; return }
  $src = '\\tsclient\D\Claude\Projects\RTM View Shell'; $ops = 'C:\RTMView-Ops'; $m = "$src\.coord\measure\namebackfill-1106"
  $stamp = Get-Date -Format yyyyMMdd_HHmmss; $work = "$ops\output\namebackfill_REAL_$stamp"
  $inbox = "$src\.coord\inbox\devops.md"; if (-not (Test-Path -LiteralPath $inbox)) { Write-Host 'STOP: inbox not readable' -ForegroundColor Red; return }
  $headings = @(Select-String -LiteralPath $inbox -Pattern '^## ' -Encoding UTF8 | ForEach-Object { $_.Line }); $blessAt = -1
  for ($index = 0; $index -lt $headings.Count; $index++) { $line = $headings[$index]; if ($line.Contains('coordinator-') -and $line.Contains('-> devops-1106') -and $line.Contains('PASS') -and $line.ToUpper().Contains('BACKFILL') -and $line.Contains('commit=true')) { $blessAt = $index } }
  Tick "blessing (coordinator-* -> devops-1106, PASS + BACKFILL + commit=true, the last such) at letter $blessAt of $($headings.Count) (-1 = none)"; if ($blessAt -lt 0) { Write-Host 'STOP: no blessing - nothing written' -ForegroundColor Red; return }
  $stopWords = @('СТОП','STOP','ОТМЕН','ОТЗЫВ'); $stopHits = 0
  for ($index = $blessAt + 1; $index -lt $headings.Count; $index++) { if (@($stopWords | Where-Object { $headings[$index].ToUpper().Contains($_) }).Count) { $stopHits++; Tick ("  stop word after the blessing: " + $headings[$index].Substring(0, [math]::Min(140, $headings[$index].Length))) } }
  $control = @($stopWords | Where-Object { 'ПРОБА: СТОП ЗАПИСИ'.Contains($_) }).Count; Tick "stop hits after the blessing $stopHits (expected 0) ; POSCTL on a known stop line $control (must be >= 1)"
  if ($control -lt 1 -or $stopHits -gt 0) { Write-Host 'STOP: a stop is standing (or the stop reader is blind) - nothing written' -ForegroundColor Red; return }
  if (Test-Path -LiteralPath $work) { Write-Host 'STOP: the run folder already exists' -ForegroundColor Red; return }
  New-Item -ItemType Directory -Force -Path "$ops\probes", $work | Out-Null
  $inFolder = @(Get-ChildItem -LiteralPath $work -Force).Count; Tick "NEW run folder $work : entries $inFolder (expected 0) ; snapshot_before.csv present: $(Test-Path "$work\snapshot_before.csv") (expected False)"
  if ($inFolder -ne 0 -or (Test-Path "$work\snapshot_before.csv")) { Write-Host 'STOP: run folder not empty' -ForegroundColor Red; return }
  $sqls = @( @{ Name = 'PRE.sql'; Sha = '32315626165FFCF01BEA29DD77784065FE5BFFBB8CB9D73ED3021C0B2C0AE811' }, @{ Name = 'BACKFILL.sql'; Sha = 'FBA0C257EE3879A3F206662917945DAE70F60EA793C88790FF956A8F630A6BF9' } )
  foreach ($sqlFile in $sqls) { $dest = "$ops\probes\namebackfill_" + $sqlFile.Name; Copy-Item -LiteralPath "$m\$($sqlFile.Name)" -Destination $dest -Force; Unblock-File -LiteralPath $dest
    $hash = (Get-FileHash -LiteralPath $dest -Algorithm SHA256).Hash; Tick ("{0} sha256 expected {1}... measured {2}" -f $sqlFile.Name, $sqlFile.Sha.Substring(0,8), $hash)
    if ($hash -ne $sqlFile.Sha) { Write-Host 'STOP: sha differs' -ForegroundColor Red; return } }
  $cfg = [IO.File]::ReadAllText('C:\RTMView\Shell\appsettings.json') | ConvertFrom-Json; $cs = [string]$cfg.ConnectionStrings.Default
  $port = [regex]::Match($cs, '(?i)Port\s*=\s*([^;]+)').Groups[1].Value.Trim(); $db = [regex]::Match($cs, '(?i)Database\s*=\s*([^;]+)').Groups[1].Value.Trim(); $user = [regex]::Match($cs, '(?i)(?:Username|User ID|UserId|User)\s*=\s*([^;]+)').Groups[1].Value.Trim()
  Tick "db localhost port $port database $db user $user (expected 5433 rtmviewdb)"; if ($port -ne '5433') { Write-Host 'STOP' -ForegroundColor Red; return }
  $psql = 'C:\Program Files\PostgreSQL\18\bin\psql.exe'
  $realArgs = @('-h','localhost','-p',$port,'-d',$db,'-U',$user,'-X','-v','commit=true','-f',"$ops\probes\namebackfill_BACKFILL.sql",'-o',"$work\backfill_real.out")
  Tick ("REAL RUN mode argument: " + (@($realArgs | Where-Object { $_ -like 'commit=*' }) -join ',') + " (expected commit=true) - THIS WRITES to hist_agent_intervals")
  $sec = Read-Host -Prompt 'DB password' -AsSecureString; $bstr = [Runtime.InteropServices.Marshal]::SecureStringToBSTR($sec)
  try { $env:PGPASSWORD = [Runtime.InteropServices.Marshal]::PtrToStringAuto($bstr); $env:PGCLIENTENCODING = 'UTF8'
    Tick '1/3 PRE (read only)'; $env:PGOPTIONS = '-c default_transaction_read_only=on'
    Start-Process -FilePath $psql -WorkingDirectory $work -NoNewWindow -Wait -RedirectStandardError "$work\pre1.err" -ArgumentList @('-h','localhost','-p',$port,'-d',$db,'-U',$user,'-X','-f',"$ops\probes\namebackfill_PRE.sql",'-o',"$work\pre1.out")
    Tick '2/3 BACKFILL commit=true (one transaction, checks, COMMIT only if every check passed)'; $env:PGOPTIONS = $null
    Start-Process -FilePath $psql -WorkingDirectory $work -NoNewWindow -Wait -RedirectStandardError "$work\backfill_real.err" -ArgumentList $realArgs
    $snap = "$work\snapshot_before.csv"
    if (Test-Path -LiteralPath $snap) { Tick ("snapshot_before.csv : sha256 {0} ; lines {1} (header + target rows)" -f (Get-FileHash -LiteralPath $snap -Algorithm SHA256).Hash, @(Get-Content -LiteralPath $snap).Count) } else { Tick 'snapshot_before.csv : ABSENT' }
    Tick '3/3 PRE again (read only) - expected rows_unnamed 0 (plus any rows the aggregator wrote unnamed in between - it writes names since 19:44)'; $env:PGOPTIONS = '-c default_transaction_read_only=on'
    Start-Process -FilePath $psql -WorkingDirectory $work -NoNewWindow -Wait -RedirectStandardError "$work\pre2.err" -ArgumentList @('-h','localhost','-p',$port,'-d',$db,'-U',$user,'-X','-f',"$ops\probes\namebackfill_PRE.sql",'-o',"$work\pre2.out")
  } finally { $env:PGPASSWORD = $null; $env:PGOPTIONS = $null; [Runtime.InteropServices.Marshal]::ZeroFreeBSTR($bstr) }
  $outcome = @(Get-Content "$work\backfill_real.out" -ErrorAction SilentlyContinue | Where-Object { $_ -match 'ROLLED BACK|COMMITTED' }) -join ' '
  $checks = @(Get-Content "$work\backfill_real.err" -ErrorAction SilentlyContinue | Where-Object { $_ -match 'ALL CHECKS PASSED|ERROR' }) -join ' '
  Tick "REAL-run outcome: [$outcome] (expected COMMITTED) ; checks: [$checks] (expected ALL CHECKS PASSED, no ERROR)"
  $pre1 = @(Get-Content "$work\pre1.out" | Where-Object { $_ -match '^\s*R[01] ' }); $pre2 = @(Get-Content "$work\pre2.out" | Where-Object { $_ -match '^\s*R[01] ' })
  Tick "PRE 1/3: $($pre1 -join ' || ')"; Tick "PRE 3/3: $($pre2 -join ' || ')"; Tick "KEEP THIS FOLDER: $work - ROLLBACK.sql runs from it, against the snapshot sha printed above"
  foreach ($name in @('pre1.err','backfill_real.err','pre2.err')) { Tick ("{0}: ERROR lines {1}" -f $name, @(Get-Content "$work\$name" -ErrorAction SilentlyContinue | Where-Object { $_ -match '\bERROR\b' }).Count) }
  $backDir = "$m\run_REAL_$stamp"; New-Item -ItemType Directory -Force -Path $backDir | Out-Null
  foreach ($file in @(Get-ChildItem -LiteralPath $work -File)) { Copy-Item -LiteralPath $file.FullName -Destination "$backDir\$($file.Name)" -Force; Write-Host ("copied {0}  sha equal: {1}" -f $file.Name, ((Get-FileHash $file.FullName).Hash -eq (Get-FileHash "$backDir\$($file.Name)").Hash)) }
}
