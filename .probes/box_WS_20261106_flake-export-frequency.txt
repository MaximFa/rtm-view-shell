& {
  function Tick($text) { Write-Host ("[{0}] {1}" -f (Get-Date -Format 'HH:mm:ss'), $text) -ForegroundColor Cyan }
  if ($env:COMPUTERNAME -eq 'RTM') { Write-Host 'STOP: this is server 234' -ForegroundColor Red; return }
  $repo = 'D:\Claude\Projects\RTM-design'; $out = 'D:\Claude\Projects\RTM View Shell\.coord\measure\flake-export-1106'; New-Item -ItemType Directory -Force -Path $out | Out-Null
  Start-Transcript -LiteralPath (Join-Path $out ('flake_{0}.txt' -f (Get-Date -Format 'yyyyMMdd_HHmmss'))) | Out-Null
  $oldClones = @(Get-ChildItem -LiteralPath 'D:\RTMView-Ops\clones' -Directory -ErrorAction SilentlyContinue); foreach ($oldClone in $oldClones) { Remove-Item -LiteralPath $oldClone.FullName -Recurse -Force -ErrorAction SilentlyContinue }
  Tick ("old clones before " + $oldClones.Count + " ; left " + @(Get-ChildItem -LiteralPath 'D:\RTMView-Ops\clones' -Directory -ErrorAction SilentlyContinue).Count + " (expected 0)")
  $clone = Join-Path 'D:\RTMView-Ops\clones' ('flake_' + (Get-Date -Format 'HHmmss'))
  try {
    Tick 'PR234-TEST-EXPORT-XLSX-DUPLICATE-RELS-FLAKE-01 frequency (coordinator 19:45:24Z): clean clone of v4 b7ac9dd, one build, ExportEndOfShiftTests x20, then the whole Unit set x10. Station only, nothing committed.'
    $head = (git -C $repo rev-parse v4).Trim(); Tick "v4 head $($head.Substring(0,7)) (expected b7ac9dd)"; if (-not $head.StartsWith('b7ac9dd')) { Write-Host 'STOP: head moved' -ForegroundColor Red; return }
    git clone -q --branch v4 $repo $clone; Tick "clone $clone at $((git -C $clone rev-parse --short HEAD).Trim()) ; dirty $(@(git -C $clone status --porcelain -uall).Count) (expected 0)"
    $proj = Join-Path $clone 'tests\CcDashboard.Tests.Unit\CcDashboard.Tests.Unit.csproj'
    $buildText = dotnet build $proj -c Release --no-incremental 2>&1 | Out-String; Tick "build exit $LASTEXITCODE (expected 0)"; if ($LASTEXITCODE -ne 0) { Write-Host 'STOP: build' -ForegroundColor Red; return }
    function Run-Once([string]$label, [string]$filter, [int]$index) {
      $testArgs = @('test', $proj, '-c', 'Release', '--no-build'); if ($filter) { $testArgs += @('--filter', $filter) }
      $text = & dotnet @testArgs 2>&1 | Out-String
      $summary = [regex]::Match($text, '(Passed|Failed)!\s+-\s+Failed:\s+(\d+), Passed:\s+(\d+), Skipped:\s+(\d+), Total:\s+(\d+)')
      $failedCount = $(if ($summary.Success) { [int]$summary.Groups[2].Value } else { -1 }); $total = $(if ($summary.Success) { [int]$summary.Groups[5].Value } else { -1 })
      $line = ("{0} #{1,2}: total {2} ; failed {3}" -f $label, $index, $total, $failedCount)
      if ($failedCount -ne 0) { $names = @(($text -split "`r?`n") | Where-Object { $_ -match '^\s+Failed\s+\S' } | ForEach-Object { ($_.Trim() -replace '^Failed\s+', '') -replace '\s+\[[^\]]*\]\s*$', '' })
        $lines = $text -split "`r?`n"; $at = [Array]::FindIndex($lines, [Predicate[string]]{ param($candidate) $candidate -match 'Error Message:' })
        $first = $(if ($at -ge 0) { $lines[$at + 1].Trim() } else { '(no Error Message line)' })
        $line += (" ; red: " + ($names -join ', ') + " ; first error: " + $first)
        [IO.File]::WriteAllText((Join-Path $out ("testlog_{0}_{1:00}.txt" -f $label, $index)), $text) }
      Tick $line; return [pscustomobject]@{ Total = $total; Failed = $failedCount } }
    $classRuns = @(); for ($index = 1; $index -le 20; $index++) { $classRuns += Run-Once 'class' 'FullyQualifiedName~HistoricalReports.ExportEndOfShiftTests' $index }
    $unitRuns = @(); for ($index = 1; $index -le 10; $index++) { $unitRuns += Run-Once 'unit' '' $index }
    $classRed = @($classRuns | Where-Object { $_.Failed -ne 0 }).Count; $unitRed = @($unitRuns | Where-Object { $_.Failed -ne 0 }).Count
    $classTotals = (@($classRuns | ForEach-Object { $_.Total } | Sort-Object -Unique) -join ','); $unitTotals = (@($unitRuns | ForEach-Object { $_.Total } | Sort-Object -Unique) -join ',')
    Tick ("SUMMARY: ExportEndOfShiftTests red runs {0} of 20 (case totals seen: {1}) ; whole Unit red runs {2} of 10 (totals seen: {3}, expected 685)" -f $classRed, $classTotals, $unitRed, $unitTotals)
    Tick 'NOTE: the red of 21:34 was inside dotnet test of the whole solution (five test projects); this box runs the Unit project alone, as asked - a 0 here does not rule out a cross-project condition.'
  } finally {
    if (Test-Path -LiteralPath $clone) { Remove-Item -LiteralPath $clone -Recurse -Force -ErrorAction SilentlyContinue; Tick "clone removed: $(-not (Test-Path -LiteralPath $clone))" }
    Stop-Transcript | Out-Null }
}
